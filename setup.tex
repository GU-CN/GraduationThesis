% 中文封面标签：固定 6em，冒号贴在标签盒右侧，保证各行冒号对齐
% 不用模板内部私有宏，避免 Overleaf / 不同 TeX 版本下失效
\newcommand{\CoverLabelZh}[1]{%
  \makebox[6em][l]{\heiti #1}：%
}

% 中文封面：单层表格；续行不输出冒号；姓名/导师/学院内容同一左缘
\newcommand{\CoverInfoZh}{%
  \begingroup
  \centering
  \zihao{4}%
  \renewcommand{\arraystretch}{1.2}%
  \begin{tabular}{@{}l@{\hspace{0.8em}}l@{\hspace{3em}}l@{}}
    \CoverLabelZh{姓名} & \TeamMemberOneZh   & \TeamMemberOneID   \\
                        & \TeamMemberTwoZh   & \TeamMemberTwoID   \\
                        & \TeamMemberThreeZh & \TeamMemberThreeID \\
                        & \TeamMemberFourZh  & \TeamMemberFourID  \\
                        & \TeamMemberFiveZh  & \TeamMemberFiveID  \\
    \CoverLabelZh{导师} & \ThesisMentorZh & \\
    \CoverLabelZh{指导教师} & \ThesisInstructorZh & \\
    \CoverLabelZh{学院} & \ThesisDepartmentZh & \\
    \CoverLabelZh{专业名称} & \ThesisMajorZh & \\
    \CoverLabelZh{申请学位层次} & \ThesisDegreeZh &
  \end{tabular}%
  \endgroup
}
